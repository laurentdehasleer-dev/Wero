amount.addEventListener("input", (e) => { // Récupérer uniquement les chiffres et le point let value = e.target.value.replace(/[^0-9.]/g, ""); // Corriger le format : max 2 décimales if (value.includes('.')) { let parts = value.split('.'); // Garder seulement le premier point value = parts[0] + '.' + parts.slice(1).join('').substring(0, 2); } else if (value.length > 0) { // Rien à faire, laisser comme ça } // Afficher avec le format "XX.XX €" if (value) { // S'assurer qu'il y a 2 décimales let [euros, cents] = value.split('.'); cents = cents || '00'; cents = cents.padEnd(2, '0').substring(0, 2); e.target.value = euros + '.' + cents + ' €'; } else { e.target.value = ''; } });
